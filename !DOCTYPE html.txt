    // Audio Toggle
    const musicBtn = document.getElementById('musicBtn');
    const bgMusic = document.getElementById('bgMusic');
    let isPlaying = false;

    musicBtn.addEventListener('click', () => {
      if (isPlaying) {
        bgMusic.pause();
      } else {
        bgMusic.play().catch(e => console.log(e));
      }
      isPlaying = !isPlaying;
    });

    // 3D Envelope Animation
    const envelopeBtn = document.getElementById('envelopeBtn');
    const envelopeFlap = document.getElementById('envelopeFlap');
    const waxSeal = document.getElementById('waxSeal');
    const envelopeScreen = document.getElementById('envelopeScreen');

    envelopeBtn.addEventListener('click', () => {
      // Step 1: Hide seal
      waxSeal.style.transition = 'transform 0.4s ease, opacity 0.4s ease';
      waxSeal.style.transform = 'translateX(-50%) scale(0)';
      waxSeal.style.opacity = '0';

      // Step 2: Open Flap
      setTimeout(() => {
        envelopeFlap.style.transform = 'rotateX(180deg)';
      }, 300);

      // Step 3: Play Music and Fade Envelope Out
      setTimeout(() => {
        bgMusic.play().catch(e => console.log('Autoplay blocked', e));
        isPlaying = true;
        
        envelopeScreen.style.transition = 'opacity 1s ease, visibility 1s';
        envelopeScreen.style.opacity = '0';
        envelopeScreen.style.visibility = 'hidden';
      }, 1200);
    });
